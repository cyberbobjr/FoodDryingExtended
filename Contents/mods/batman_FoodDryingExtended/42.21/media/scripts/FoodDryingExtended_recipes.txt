    /* Large fruits are sliced before drying (vanilla slicing pattern: the nutrition is split between the slices). */

    craftRecipe SliceLargeFruit
    {
        timedAction = SliceFood_Surface,
        time = 50,
        Tags = AnySurfaceCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 tags[base:sharpknife;base:meatcleaver] mode:keep flags[SharpnessCheck],
            item 1 [Base.Pineapple;Base.Mango;Base.Grapefruit] flags[InheritFood;ItemCount] mappers[slices],
        }
        outputs
        {
            item 4 mapper:slices,
        }
        itemMapper slices
        {
            FoodDryingExtended.PineappleSlice = Base.Pineapple,
            FoodDryingExtended.MangoSlice = Base.Mango,
            FoodDryingExtended.GrapefruitSlice = Base.Grapefruit,
        }
    }
